<body>
    <div id="app-container" class="day-mode">
        <div id="splash-screen" class="screen flex-center">
            <h1 class="splash-text">RAMAN PRESENTS</h1>
        </div>

        <div id="start-menu" class="screen flex-center hidden">
            <h1 class="title-text">LANE RUNNER</h1>
            <div class="panel">
                <p class="high-score">High Score: <span id="menu-high-score">0</span></p>
                <button id="start-btn" class="ui-btn primary-btn">Start Game</button>
                <button id="theme-btn" class="ui-btn secondary-btn">Toggle Night Mode</button>
            </div>
        </div>

        <div id="game-ui" class="screen hidden">
            <div id="score-hud">Score: <span id="game-score">0</span> | Coins: <span id="game-coins">0</span></div>
            <canvas id="game-canvas"></canvas>
            <div id="mobile-controls">
                <button id="btn-left" class="control-btn">⬅️ Left</button>
                <button id="btn-right" class="control-btn">Right ➡️</button>
            </div>
        </div>

        <div id="game-over-screen" class="screen flex-center hidden">
            <div class="panel">
                <h2 class="game-over-text">Game Over!</h2>
                <p class="score-text">Final Score: <span id="final-score">0</span></p>
                <button id="restart-btn" class="ui-btn primary-btn">Restart</button>
                <button id="home-btn" class="ui-btn secondary-btn">Main Menu</button>
            </div>

    <script>
        // --- DOM Elements ---
        const appContainer = document.getElementById('app-container');
        const splashScreen = document.getElementById('splash-screen');
        const startMenu = document.getElementById('start-menu');
        const gameUI = document.getElementById('game-ui');
        const gameOverScreen = document.getElementById('game-over-screen');

        const canvas = document.getElementById('game-canvas');
        const ctx = canvas.getContext('2d');

        const menuHighScoreEl = document.getElementById('menu-high-score');
        const gameScoreEl = document.getElementById('game-score');
        const gameCoinsEl = document.getElementById('game-coins');
        const finalScoreEl = document.getElementById('final-score');

        // Buttons
        const startBtn = document.getElementById('start-btn');
        const themeBtn = document.getElementById('theme-btn');
        const restartBtn = document.getElementById('restart-btn');
        const homeBtn = document.getElementById('home-btn');
        const btnLeft = document.getElementById('btn-left');
        const btnRight = document.getElementById('btn-right');

        // --- Game Variables ---
        let animationId;
        let isPlaying = false;
        let isNightMode = false;

        // Game State
        let score = 0;
        let coinsCollected = 0;
        let gameSpeed = 5;
        let baseSpeed = 5;
        let frameCount = 0;
        let highScore = localStorage.getItem('laneRunnerHighScore') || 0;
        menuHighScoreEl.textContent = Math.floor(highScore);

        // Sizing
        const gameWidth = 400;
        const gameHeight = 800;
        canvas.width = gameWidth;
        canvas.height = gameHeight;

        // Lane Configuration
        const lanes = [gameWidth * 0.2, gameWidth * 0.5, gameWidth * 0.8]; // Left, Center, Right X coordinates
        let currentLane = 1; // 0, 1, 2

        // Entities
        let player = { x: lanes[1], y: gameHeight - 120, size: 40 };
        let obstacles = [];
        let coins = [];
        let roadLines = [];

        // --- Audio Effects (Synthesized to avoid external assets) ---
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        function playCoinSound() {
            if (audioCtx.state === 'suspended') audioCtx.resume();
            const oscillator = audioCtx.createOscillator();
            const gainNode = audioCtx.createGain();
            
            oscillator.type = 'sine';
            oscillator.frequency.setValueAtTime(800, audioCtx.currentTime);
            oscillator.frequency.exponentialRampToValueAtTime(1600, audioCtx.currentTime + 0.1);
            
            gainNode.gain.setValueAtTime(0.3, audioCtx.currentTime);
            gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
            
            oscillator.connect(gainNode);
            gainNode.connect(audioCtx.destination);
            oscillator.start();
            oscillator.stop(audioCtx.currentTime + 0.1);
        }

        // --- Initialization & Setup ---
        setTimeout(() => {
            splashScreen.classList.add('hidden');
            startMenu.classList.remove('hidden');
        }, 2000);

        themeBtn.addEventListener('click', () => {
            isNightMode = !isNightMode;
            appContainer.className = isNightMode ? 'night-mode' : 'day-mode';
        });

        startBtn.addEventListener('click', startGame);
        restartBtn.addEventListener('click', startGame);
        homeBtn.addEventListener('click', () => {
            gameOverScreen.classList.add('hidden');
            startMenu.classList.remove('hidden');
            menuHighScoreEl.textContent = Math.floor(highScore);
        });

        // --- Input Handling ---
        window.addEventListener('keydown', (e) => {
            if (!isPlaying) return;
            if (e.key === 'ArrowLeft' && currentLane > 0) currentLane--;
            if (e.key === 'ArrowRight' && currentLane < 2) currentLane++;
        });

        // Add touch event listeners for better mobile response, fallback to click
        btnLeft.addEventListener('touchstart', (e) => {
            e.preventDefault(); // Prevent default touch behavior like double-tap zoom
            if (isPlaying && currentLane > 0) currentLane--;
        });
        btnLeft.addEventListener('click', () => {
            if (isPlaying && currentLane > 0) currentLane--;
        });

        btnRight.addEventListener('touchstart', (e) => {
            e.preventDefault();
            if (isPlaying && currentLane < 2) currentLane++;
        });
        btnRight.addEventListener('click', () => {
            if (isPlaying && currentLane < 2) currentLane++;
        });

        // --- Game Logic ---
        function startGame() {
            startMenu.classList.add('hidden');
            gameOverScreen.classList.add('hidden');
            gameUI.classList.remove('hidden');
            
            // Reset Stats
            score = 0;
            coinsCollected = 0;
            gameSpeed = baseSpeed;
            frameCount = 0;
            currentLane = 1;
            obstacles = [];
            coins = [];
            roadLines = [];
            isPlaying = true;

            // Initial Road Lines
            for(let i=0; i<gameHeight; i+=60) {
                roadLines.push(i);
            }

            if (audioCtx.state === 'suspended') audioCtx.resume();
            gameLoop();
        }

        function spawnEntities() {
            frameCount++;
            
            // Increase speed gradually
            if (frameCount % 60 === 0) gameSpeed += 0.05;

            // Spawn Obstacle
            if (frameCount % Math.max(30, Math.floor(100 - gameSpeed*2)) === 0) {
                const laneIdx = Math.floor(Math.random() * 3);
                const types = ['🚧', '📦', '🛢️'];
                obstacles.push({
                    x: lanes[laneIdx],
                    y: -50,
                    type: types[Math.floor(Math.random() * types.length)],
                    size: 45
                });
            }

            // Spawn Coins
            if (frameCount % 45 === 0) {
                const laneIdx = Math.floor(Math.random() * 3);
                // Avoid overlapping with latest obstacle
                const lastObstacle = obstacles[obstacles.length-1];
                if (!lastObstacle || lastObstacle.x !== lanes[laneIdx] || lastObstacle.y > 100) {
                    coins.push({
                        x: lanes[laneIdx],
                        y: -50,
                        size: 30,
                        collected: false
                    });
                }
            }
        }

        function update() {
            // Smooth player lane transition
            player.x += (lanes[currentLane] - player.x) * 0.2;
            
            // Update Score
            score += gameSpeed * 0.02;
            gameScoreEl.textContent = Math.floor(score);
            gameCoinsEl.textContent = coinsCollected;

            // Update Road Lines
            for (let i = 0; i < roadLines.length; i++) {
                roadLines[i] += gameSpeed;
                if (roadLines[i] > gameHeight) roadLines[i] = -60;
            }

            // Update Obstacles & Collision
            for (let i = obstacles.length - 1; i >= 0; i--) {
                obstacles[i].y += gameSpeed;
                
                // Collision Detection (Circle approximation)
                const dx = player.x - obstacles[i].x;
                const dy = player.y - obstacles[i].y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance < (player.size/2 + obstacles[i].size/2 - 5)) {
                    endGame();
                }
                
                // Remove off-screen
                if (obstacles[i].y > gameHeight + 100) obstacles.splice(i, 1);
            }

            // Update Coins
            for (let i = coins.length - 1; i >= 0; i--) {
                coins[i].y += gameSpeed;
                
                if (!coins[i].collected) {
                    const dx = player.x - coins[i].x;
                    const dy = player.y - coins[i].y;
                    const distance = Math.sqrt(dx * dx + dy * dy);
                    
                    if (distance < (player.size/2 + coins[i].size/2 + 10)) {
                        coins[i].collected = true;
                        coinsCollected++;
                        score += 50; // Bonus points
                        playCoinSound();
                        coins.splice(i, 1);
                        continue;
                    }
                }
                
                if (coins[i].y > gameHeight + 100) coins.splice(i, 1);
            }
        }

        function draw() {
            // Clear & draw background
            ctx.fillStyle = isNightMode ? '#222' : '#555'; // Road color
            ctx.fillRect(0, 0, gameWidth, gameHeight);

            // Draw Lane Markings
            ctx.fillStyle = isNightMode ? '#AA0' : '#FFF';
            const laneWidth = gameWidth / 3;
            
            for (let lineY of roadLines) {
                ctx.fillRect(laneWidth - 5, lineY, 10, 40);
                ctx.fillRect(laneWidth * 2 - 5, lineY, 10, 40);
            }

            // Draw Coins
            ctx.font = '30px Arial';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            coins.forEach(coin => {
                ctx.fillText('🪙', coin.x, coin.y);
            });

            // Draw Obstacles
            ctx.font = '45px Arial';
            obstacles.forEach(obs => {
                ctx.fillText(obs.type, obs.x, obs.y);
            });

            // Draw Player (Girl character emoji)
            ctx.font = '50px Arial';
            ctx.fillText('👧', player.x, player.y);
        }

        function gameLoop() {
            if (!isPlaying) return;
            
            spawnEntities();
            update();
            draw();
            
            animationId = requestAnimationFrame(gameLoop);
        }

        function endGame() {
            isPlaying = false;
            cancelAnimationFrame(animationId);
            
            if (score > highScore) {
                highScore = score;
                localStorage.setItem('laneRunnerHighScore', highScore);
            }
            
            gameUI.classList.add('hidden');
            gameOverScreen.classList.remove('hidden');
            finalScoreEl.textContent = Math.floor(score);
        }
    </script>
</body>
