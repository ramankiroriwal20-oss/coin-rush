<script>

setTimeout(()=>{
document.getElementById("splash").style.display="none";
},2000);

let player=document.getElementById("player");
let score=0;
let playerX=window.innerWidth/2;

function updatePlayer(){
player.style.left=playerX+"px";
}

document.addEventListener("keydown",(e)=>{
if(e.key==="ArrowLeft") playerX-=25;
if(e.key==="ArrowRight") playerX+=25;

if(playerX<20) playerX=20;
if(playerX>window.innerWidth-20) playerX=window.innerWidth-20;

updatePlayer();
});

document.addEventListener("touchmove",(e)=>{
playerX=e.touches[0].clientX;
updatePlayer();
});

function createCoin(){

let coin=document.createElement("div");
coin.className="coin";
coin.innerHTML="🪙";

let x=Math.random()*(window.innerWidth-40);

coin.style.left=x+"px";
coin.style.top="0px";

document.body.appendChild(coin);

let y=0;

let fall=setInterval(()=>{

y+=5;
coin.style.top=y+"px";

let playerRect=player.getBoundingClientRect();
let coinRect=coin.getBoundingClientRect();

if(
coinRect.bottom>playerRect.top &&
coinRect.top<playerRect.bottom &&
coinRect.left<playerRect.right &&
coinRect.right>playerRect.left
){
score++;
document.getElementById("score").innerHTML="Score: "+score;
coin.remove();
clearInterval(fall);
}

if(y>window.innerHeight){
coin.remove();
clearInterval(fall);
}

},20);
}

setInterval(createCoin,1000);

</script>
