    // Game mechanics

    SPEED_INCREMENT: 0.001,
    
    // Obstacles
    OBSTACLE_MIN_WIDTH: 30,
    OBSTACLE_MAX_WIDTH: 50,
    OBSTACLE_MIN_HEIGHT: 40,
    OBSTACLE_MAX_HEIGHT: 80,
    OBSTACLE_MIN_GAP: 800,
    OBSTACLE_MAX_GAP: 1500,

// ============================================
// Player Class
// ============================================
class Player {
    constructor(x, y) {
        this.x = x;
        this.y = y;
        this.width = CONFIG.PLAYER_WIDTH;
        this.height = CONFIG.PLAYER_HEIGHT;
        this.velocityY = 0;
        this.isJumping = false;
        this.runFrame = 0;
        this.frameCount = 0;
    }

    /**
     * Make the player jump
     */
    jump() {
        if (!this.isJumping) {
            this.velocityY = CONFIG.JUMP_FORCE;
            this.isJumping = true;
            this.createJumpParticles();
        }
    }

    /**
     * Create particle effect when jumping
     */
    createJumpParticles() {
        for (let i = 0; i < 8; i++) {
            particles.push(new Particle(
                this.x + this.width / 2,
                this.y + this.height,
                (Math.random() - 0.5) * 4,
                Math.random() * 3 + 1,
                CONFIG.PLAYER_COLOR
            ));
        }
    }

    /**
     * Update player physics
     */
    update() {
        // Apply gravity
        this.velocityY += CONFIG.GRAVITY;
        this.y += this.velocityY;

        // Ground collision
        if (this.y + this.height >= groundY) {
            this.y = groundY - this.height;
            this.velocityY = 0;
            this.isJumping = false;
        }

        // Running animation
        this.frameCount++;
        if (this.frameCount % 6 === 0) {
            this.runFrame = (this.runFrame + 1) % 4;
        }
    }

    /**
     * Draw the player
     */
    draw() {
        ctx.save();

        // Player body with glow
        ctx.shadowColor = CONFIG.PLAYER_COLOR;
        ctx.shadowBlur = 20;
        ctx.fillStyle = CONFIG.PLAYER_COLOR;

        // Draw stylized running character
        const legOffset = Math.sin(this.runFrame * Math.PI / 2) * 8;
        
        // Body
        ctx.fillRect(this.x + 10, this.y + 10, 20, 30);
        
        // Head
        ctx.beginPath();
        ctx.arc(this.x + 20, this.y + 5, 12, 0, Math.PI * 2);
        ctx.fill();

        // Legs (animated)
        ctx.fillRect(this.x + 8, this.y + 40, 8, 20 + legOffset);
        ctx.fillRect(this.x + 24, this.y + 40, 8, 20 - legOffset);

        // Arms (animated)
        ctx.fillRect(this.x + 2, this.y + 15 - legOffset / 2, 8, 15);
        ctx.fillRect(this.x + 30, this.y + 15 + legOffset / 2, 8, 15);

        // Eye visor effect
        ctx.fillStyle = '#ffffff';
        ctx.shadowBlur = 10;
        ctx.fillRect(this.x + 14, this.y + 2, 12, 4);

        ctx.restore();
    }

    /**
     * Get hitbox for collision detection (slightly smaller for fairness)
     */
    getHitbox() {
        return {
            x: this.x + 5,
            y: this.y + 5,
            width: this.width - 10,
            height: this.height - 10
        };
    }
}

// ============================================
// Obstacle Class
// ============================================
class Obstacle {
    constructor(x) {
        this.x = x;
        this.width = randomRange(CONFIG.OBSTACLE_MIN_WIDTH, CONFIG.OBSTACLE_MAX_WIDTH);
        this.height = randomRange(CONFIG.OBSTACLE_MIN_HEIGHT, CONFIG.OBSTACLE_MAX_HEIGHT);
        this.y = groundY - this.height;
        this.passed = false;
    }

    /**
     * Update obstacle position
     */
    update() {
        this.x -= gameState.speed;
    }

    /**
     * Draw the obstacle
     */
    draw() {
        ctx.save();

        // Glowing effect
        ctx.shadowColor = CONFIG.OBSTACLE_COLOR;
        ctx.shadowBlur = 15;

        // Main obstacle body
        ctx.fillStyle = CONFIG.OBSTACLE_COLOR;
        ctx.fillRect(this.x, this.y, this.width, this.height);

        // Top highlight
        ctx.fillStyle = '#ff3366';
        ctx.fillRect(this.x, this.y, this.width, 5);

        // Side accent
        ctx.fillStyle = '#cc0033';
        ctx.fillRect(this.x + this.width - 5, this.y, 5, this.height);

        // Warning stripes
        ctx.fillStyle = '#0a0a0f';
        const stripeCount = Math.floor(this.height / 15);
        for (let i = 0; i < stripeCount; i++) {
            if (i % 2 === 0) {
                ctx.fillRect(this.x + 5, this.y + i * 15 + 5, this.width - 10, 5);
            }
        }

        ctx.restore();
    }

    /**
     * Check if obstacle is off screen
     */
    isOffScreen() {
        return this.x + this.width < 0;
    }

    /**
     * Get hitbox for collision
     */
    getHitbox() {
        return {
            x: this.x + 3,
            y: this.y + 3,
            width: this.width - 6,
            height: this.height - 6
        };
    }
}

// ============================================
// Coin Class
// ============================================
class Coin {
    constructor(x, y) {
        this.x = x;
        this.y = y;
        this.size = CONFIG.COIN_SIZE;
        this.collected = false;
        this.rotation = 0;
        this.bobOffset = Math.random() * Math.PI * 2;
    }

    /**
     * Update coin position and animation
     */
    update() {
        this.x -= gameState.speed;
        this.rotation += 0.1;
        this.bobOffset += 0.08;
    }

    /**
     * Draw the coin
     */
    draw() {
        if (this.collected) return;

        ctx.save();
        
        const bobY = this.y + Math.sin(this.bobOffset) * 5;
        const scaleX = Math.cos(this.rotation);

        ctx.translate(this.x, bobY);
        ctx.scale(Math.abs(scaleX) * 0.5 + 0.5, 1);

        // Glow effect
        ctx.shadowColor = CONFIG.COIN_COLOR;
        ctx.shadowBlur = 20;

        // Coin body
        ctx.fillStyle = CONFIG.COIN_COLOR;
        ctx.beginPath();
        ctx.arc(0, 0, this.size / 2, 0, Math.PI * 2);
        ctx.fill();

        // Inner highlight
        ctx.fillStyle = '#ffee88';
        ctx.beginPath();
        ctx.arc(-3, -3, this.size / 4, 0, Math.PI * 2);
        ctx.fill();

        ctx.restore();
    }

    /**
     * Check if coin is off screen
     */
    isOffScreen() {
        return this.x + this.size < 0;
    }

    /**
     * Get hitbox for collection
     */
    getHitbox() {
        return {
            x: this.x - this.size / 2,
            y: this.y - this.size / 2,
            width: this.size,
            height: this.size
        };
    }
}

// ============================================
// Particle Class (for effects)
// ============================================
class Particle {
    constructor(x, y, vx, vy, color) {
        this.x = x;
        this.y = y;
        this.vx = vx;
        this.vy = vy;
        this.color = color;
        this.life = 1;
        this.decay = 0.02 + Math.random() * 0.02;
        this.size = 3 + Math.random() * 4;
    }

    /**
     * Update particle
     */
    update() {
        this.x += this.vx;
        this.y += this.vy;
        this.vy += 0.1; // Mini gravity
        this.life -= this.decay;
    }

    /**
     * Draw particle
     */
    draw() {
        if (this.life <= 0) return;

        ctx.save();
        ctx.globalAlpha = this.life;
        ctx.fillStyle = this.color;
        ctx.shadowColor = this.color;
        ctx.shadowBlur = 10;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size * this.life, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
    }

    /**
     * Check if particle is dead
     */
    isDead() {
        return this.life <= 0;
    }
}

/**
 * Generate random number in range
 */
function randomRange(min, max) {
    return Math.random() * (max - min) + min;
}

/**
 * Check collision between two rectangles
 */
function checkCollision(rect1, rect2) {
    return rect1.x < rect2.x + rect2.width &&
           rect1.x + rect1.width > rect2.x &&
           rect1.y < rect2.y + rect2.height &&
           rect1.y + rect1.height > rect2.y;
}

/**
 * Load high score from localStorage
 */
function loadHighScore() {
    const saved = localStorage.getItem(CONFIG.HIGH_SCORE_KEY);
    return saved ? parseInt(saved, 10) : 0;
}

/**
 * Save high score to localStorage
 */
function saveHighScore(score) {
    localStorage.setItem(CONFIG.HIGH_SCORE_KEY, score.toString());
}

/**

 */
function setupCanvas() {
    canvasWidth = window.innerWidth;
    canvasHeight = window.innerHeight;
    canvas.width = canvasWidth;
    canvas.height = canvasHeight;
    groundY = canvasHeight - CONFIG.GROUND_HEIGHT;
}

/**
 * Initialize game objects
 */
function initGameObjects() {
    // Create player
    player = new Player(100, groundY - CONFIG.PLAYER_HEIGHT);
    
    // Clear arrays
    obstacles = [];
    coins = [];
    particles = [];
    
    // Reset game state
    gameState.score = 0;
    gameState.distance = 0;
    gameState.speed = CONFIG.INITIAL_SPEED;
    gameState.lastObstacleX = canvasWidth;
    gameState.nextObstacleGap = randomRange(CONFIG.OBSTACLE_MIN_GAP, CONFIG.OBSTACLE_MAX_GAP);
    gameState.isGameOver = false;
    gameState.isRunning = true;
}

/**
 * Setup event listeners
 */
function setupEventListeners() {
    // Keyboard controls
    document.addEventListener('keydown', (e) => {
        if (e.code === 'Space') {
            e.preventDefault();
            handleJump();
        }
    });

    // Mobile jump button
    elements.jumpBtn.addEventListener('touchstart', (e) => {
        e.preventDefault();
        handleJump();
    });

    elements.jumpBtn.addEventListener('mousedown', (e) => {
        e.preventDefault();
        handleJump();
    });

    // Restart button
    elements.restartBtn.addEventListener('click', restartGame);

    // Window resize
    window.addEventListener('resize', () => {
        setupCanvas();
        if (player) {
            player.y = Math.min(player.y, groundY - player.height);
        }
    });
}

/**
 * Handle jump action
 */
function handleJump() {
    if (gameState.isRunning && !gameState.isGameOver && player) {
        player.jump();
    }
}

// Game Loop Functions
// ============================================

/**
 * Spawn obstacles
 */
function spawnObstacles() {
    const lastObstacle = obstacles[obstacles.length - 1];
    const spawnX = lastObstacle ? lastObstacle.x : gameState.lastObstacleX;

    if (canvasWidth - spawnX >= gameState.nextObstacleGap || obstacles.length === 0) {
        const newObstacle = new Obstacle(canvasWidth + 50);
        obstacles.push(newObstacle);

        // Spawn coin above some obstacles
        if (Math.random() < CONFIG.COIN_SPAWN_CHANCE) {
            const coinX = newObstacle.x + newObstacle.width / 2;
            const coinY = newObstacle.y - 60 - Math.random() * 40;
            coins.push(new Coin(coinX, coinY));
        }

        gameState.lastObstacleX = newObstacle.x;
        gameState.nextObstacleGap = randomRange(CONFIG.OBSTACLE_MIN_GAP, CONFIG.OBSTACLE_MAX_GAP);
        
        // Decrease gap as speed increases
        gameState.nextObstacleGap *= (CONFIG.INITIAL_SPEED / gameState.speed) * 0.8 + 0.2;
    }
}

/**
 * Update game objects
 */
function updateGameObjects() {
    // Update player
    player.update();

    // Update obstacles
    obstacles.forEach(obstacle => obstacle.update());
    obstacles = obstacles.filter(obstacle => !obstacle.isOffScreen());

    // Update coins
    coins.forEach(coin => coin.update());
    coins = coins.filter(coin => !coin.isOffScreen() && !coin.collected);

    // Update particles
    particles.forEach(particle => particle.update());
    particles = particles.filter(particle => !particle.isDead());

    // Increase speed gradually
    if (gameState.speed < CONFIG.MAX_SPEED) {
        gameState.speed += CONFIG.SPEED_INCREMENT;
    }

    // Update distance
    gameState.distance += gameState.speed * 0.1;
}

/**
 * Check collisions
 */
function checkCollisions() {
    const playerHitbox = player.getHitbox();

    // Check obstacle collisions
    for (const obstacle of obstacles) {
        if (checkCollision(playerHitbox, obstacle.getHitbox())) {
            gameOver();
            return;
        }
    }

    // Check coin collection
    for (const coin of coins) {
        if (!coin.collected && checkCollision(playerHitbox, coin.getHitbox())) {
            coin.collected = true;
            gameState.score += CONFIG.COIN_VALUE;
            
            // Create collection particles
            for (let i = 0; i < 10; i++) {
                particles.push(new Particle(
                    coin.x,
                    coin.y,
                    (Math.random() - 0.5) * 8,
                    (Math.random() - 0.5) * 8,
                    CONFIG.COIN_COLOR
                ));
            }
        }
    }
}

/**
 * Update HUD displays
 */
function updateHUD() {
    elements.scoreDisplay.textContent = Math.floor(gameState.score);
    elements.highScoreDisplay.textContent = Math.floor(gameState.highScore);
    elements.distanceDisplay.textContent = Math.floor(gameState.distance);
}

// ============================================
// Rendering Functions
// ============================================

/**
 * Draw background with parallax layers
 */
function drawBackground() {
    // Sky gradient
    const gradient = ctx.createLinearGradient(0, 0, 0, canvasHeight);
    gradient.addColorStop(0, '#0d0d1a');
    gradient.addColorStop(0.5, '#1a0a2e');
    gradient.addColorStop(1, '#0a0a0f');
    ctx.fillStyle = gradient;
    ctx.fillRect(0, 0, canvasWidth, canvasHeight);

    // Stars
    ctx.fillStyle = '#ffffff';
    for (let i = 0; i < 50; i++) {
        const x = (i * 73 + gameState.distance * 0.1) % canvasWidth;
        const y = (i * 97) % (groundY - 100);
        const size = (i % 3) + 1;
        ctx.globalAlpha = 0.3 + (i % 5) * 0.1;
        ctx.beginPath();
        ctx.arc(x, y, size, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;

    // Distant city silhouette
    ctx.fillStyle = '#151525';
    for (let i = 0; i < 15; i++) {
        const x = ((i * 120 - gameState.distance * 0.3) % (canvasWidth + 200)) - 100;
        const height = 50 + (i * 31) % 100;
        ctx.fillRect(x, groundY - height, 80, height);
    }
}

/**
 * Draw ground with grid effect
 */
function drawGround() {
    // Ground base
    ctx.fillStyle = '#0a0a15';
    ctx.fillRect(0, groundY, canvasWidth, CONFIG.GROUND_HEIGHT);

    // Neon grid lines
    ctx.strokeStyle = '#00ffff';
    ctx.lineWidth = 1;
    ctx.globalAlpha = 0.3;

    // Horizontal lines
    for (let i = 0; i < 5; i++) {
        const y = groundY + i * 20;
        ctx.beginPath();
        ctx.moveTo(0, y);
        ctx.lineTo(canvasWidth, y);
        ctx.stroke();
    }

    // Vertical grid lines with perspective
    const gridOffset = (gameState.distance * 2) % 50;
    for (let i = -1; i < canvasWidth / 50 + 2; i++) {
        const x = i * 50 - gridOffset;
        ctx.beginPath();
        ctx.moveTo(x, groundY);
        ctx.lineTo(x, groundY + CONFIG.GROUND_HEIGHT);
        ctx.stroke();
    }

    ctx.globalAlpha = 1;

    // Ground glow line
    ctx.strokeStyle = '#ff00ff';
    ctx.lineWidth = 3;
    ctx.shadowColor = '#ff00ff';
    ctx.shadowBlur = 15;
    ctx.beginPath();
    ctx.moveTo(0, groundY);
    ctx.lineTo(canvasWidth, groundY);
    ctx.stroke();
    ctx.shadowBlur = 0;
}

/**
 * Main render function
 */
function render() {
    // Clear canvas
    ctx.clearRect(0, 0, canvasWidth, canvasHeight);

    // Draw layers
    drawBackground();
    drawGround();

    // Draw game objects
    obstacles.forEach(obstacle => obstacle.draw());
    coins.forEach(coin => coin.draw());
    particles.forEach(particle => particle.draw());
    player.draw();
}

/**
 * Main game loop
 */
function gameLoop() {
    if (!gameState.isRunning || gameState.isGameOver) return;

    spawnObstacles();
    updateGameObjects();
    checkCollisions();
    updateHUD();
    render();

    requestAnimationFrame(gameLoop);
}

// ============================================
// Game State Functions
// ============================================

/**
 * Start the game
 */
function startGame() {
    gameState.highScore = loadHighScore();
    elements.highScoreDisplay.textContent = gameState.highScore;
    
    initGameObjects();
    gameLoop();
}

/**
 * Game over handler
 */
function gameOver() {
    gameState.isGameOver = true;
    gameState.isRunning = false;

    // Calculate final score with distance bonus
    const finalScore = Math.floor(gameState.score + gameState.distance);
    
    // Update high score
    if (finalScore > gameState.highScore) {
        gameState.highScore = finalScore;
        saveHighScore(finalScore);
    }

    // Update game over screen
    elements.finalScore.textContent = finalScore;
    elements.finalDistance.textContent = Math.floor(gameState.distance);
    elements.finalHighScore.textContent = gameState.highScore;

    // Show game over screen
    elements.gameOverScreen.classList.remove('hidden');
}

/**
 * Restart the game
 */
function restartGame() {
    elements.gameOverScreen.classList.add('hidden');
    startGame();
}

// ============================================
// Intro Sequence
// ============================================

/**
 * Handle intro sequence
 */
function playIntro() {
    // After intro duration, fade out and start game
    setTimeout(() => {
        elements.introScreen.classList.add('fade-out');
        elements.gameContainer.classList.add('visible');
        
        // Wait for fade animation to complete
        setTimeout(() => {
            elements.introScreen.style.display = 'none';
            startGame();
        }, 500);
    }, CONFIG.INTRO_DURATION);
}
